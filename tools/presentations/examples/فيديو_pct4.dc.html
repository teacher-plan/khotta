مدرستان

📌 ٤٠ متفوّقاً في كلٍّ منهما
المدرسة (أ)٤٠ من ٥٠ = ٨٠١٠٠ = ٨٠٪
المدرسة (ب)٤٠ من ٢٠٠ = ٢٠١٠٠ = ٢٠٪
النتيجةالعدد نفسه، والنسبة مختلفة

ماذا تلاحظ؟

📌 ٢٠٪ من ٤٠ = ٤٠٪ من ٢٠
(أ)٢٠٪ من ٤٠ = ٨
(ب)٤٠٪ من ٢٠ = ٨
(ج)٦٠٪ من ٨٠ = ٨٠٪ من ٦٠ = ٤٨

مثال ٨-٤

📌 ٢١ من ٣٠ في الرياضيات، و ٥٤ من ٧٥ في العلوم
١) الرياضيات٢١٣٠ = ٧١٠ = ٧٠٪
٢) العلوم٥٤٧٥ = ١٨٢٥ = ٧٢٪
٣) نقارن٧٢٪ − ٧٠٪ = ٢٪

هل إجابتي معقولة؟

📌 الجزء في البسط والكل في المقام
✘١٧ من ٢٥ = ٢٥١٧ = ١٤٧٪
✔١٧٢٥ = ٦٨١٠٠ = ٦٨٪

📌 نحوّل كل درجةٍ إلى نسبة

الأفضل١٧ من ٢٥

معاً: أنفق كلٌّ منهما ٢٤ ريالاً

📌 المبلغ نفسه، والكل مختلف

📌 نحوّلهما إلى نسبٍ مئوية
نقارن٣١٠ = ٣٠٪ ، ٨٢٥ = ٣٢٪

أكياس الطحين

📌 أيّ طحينٍ فيه نسبةٌ أكبر من الكربوهيدرات؟

عيادة الأطفال

📌 الأولاد ٢٠ والبنات ٣٠: نجد المجموع أولاً
الأولاد٩ من ٢٠ = ٤٥٪
البنات١٢ من ٣٠ = ٤٠٪
لاحظعدد البنات أكبر، ونسبة الأولاد أكبر

معاً: صحيحة أم خاطئة؟

📌 الأعداد شيء، والنسب شيءٌ آخر
١عدد الأولاد زائدي الكتلة أكبر ← خاطئة
٢النسبة الأكبر من الأولاد زائدو الكتلة ← صحيحة
٣عدد البنات ناقصات الكتلة أكبر ← صحيحة
٤نسبة البنات ناقصات الكتلة أكبر ← خاطئة

📌 الكمية المفقودة ÷ الكتلة الأصلية
