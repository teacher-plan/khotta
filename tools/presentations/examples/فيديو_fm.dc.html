ما الصيغة؟

📌 الصيغة قاعدةٌ رياضية توضّح العلاقة بين كميتين أو أكثر
بالكلماتمساحة المستطيل = الطول × العرض
بالحروفم = ل × ض

صيغة أم عبارة؟

📌 الصيغة تحتوي دائماً على إشارة التساوي =
عبارة جبريةل × ض
صيغةم = ل × ض

التعويض

📌 نضع العدد مكان الحرف ثم نحسب
نعوّضم = ٥ × ٤
نحسبم = ٢٠ سم²

معنى الكتابة

📌 العدد الملاصق للحرف يعني الضرب
٣ك٣ك = ٣ × ٥ = ١٥
طصطص = ٣ × ٧ = ٢١
خطّ الكسرد٤ = ٣٢ ÷ ٤ = ٨

مثال ٢-٤ (أ)

📌 أوجد قيمة س + ٣ص عندما س = ٢ ، ص = ٤
نعوّض٢ + ٣ × ٤
الضرب أولاً٢ + ١٢
ثم الجمع١٤

ترتيب العمليات

📌 أقواس ← أسس ← ضرب وقسمة ← جمع وطرح
✘ خطأ٥ + ٢ × ٣ = ٧ × ٣ = ٢١
✔ صحيح٥ + ٢ × ٣ = ٥ + ٦ = ١١

مثال ٢-٤ (ب)

📌 أوجد قيمة س(١٠ − ص) عندما س = ٤ ، ص = ٧
نعوّض٤ (١٠ − ٧)
الأقواس أولاً٤ × ٣
ثم الضرب١٢

الكسور

📌 البسط كلّه يُحسب أولاً، كأنه بين قوسين
٣٠ ÷ ح − ٢٣٠٦ − ٢ = ٥ − ٢ = ٣
(س + د) ÷ ٢١٩ + ١١٢ = ٣٠٢ = ١٥

📌 عدد الأيام في أي عدد من الأسابيع
بالكلماتعدد الأيام = ٧ × عدد الأسابيع

📌 صيغة لعدد الدقائق في أي عدد من الساعات
الصيغةد = ٦٠س
٥ ساعاتد = ٦٠ × ٥ = ٣٠٠

صيغة الأجر

📌 ع = حص + و (الساعات × أجر الساعة + العلاوة)
عدنانع = ٢٠ × ٢٢ + ٣٠
الأجر٤٤٠ + ٣٠ = ٤٧٠ ريالاً

مسألة الطهي

📌 الفرن: (٦٦ × الكتلة) + ٣٥ · الميكروويف: (٢٦ × الكتلة) + ١٥
الفرن٦٦ × ٢ + ٣٥ = ١٦٧
الميكروويف٢٦ × ٢ + ١٥ = ٦٧
