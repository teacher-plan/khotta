مساحة الأشكال المركّبة

نقسّم◆نوجد كل مساحة◆نجمع أو نطرح

ما الشكل المركّب؟

📌 شكلٌ يتكوّن من أشكالٍ بسيطة◆التجويف: فتحةٌ نطرح مساحتها
الشكل المركّبيتكوّن من تركيب أشكالٍ بسيطة
التجويفجزءٌ مقطوعٌ (فتحة) من الشكل

الخطوات

📌 نقسّم بأقلّ عددٍ من الأجزاء
١) نقسّمإلى أشكالٍ بسيطة
٢) نوجدمساحة كل شكل، وقد نوجد طولاً مفقوداً
٣) نجمع أو نطرحنطرح مساحة التجويف

مجلس جدّتي

📌 أرضيةٌ على شكل حرف L: كم متراً مربّعاً من السجاد؟
الأرضية
نقسّم ونجمع٥ × ٦ + ٣ × ٤ = ٣٠ + ١٢ = ٤٢ م²
أو: الكبير ناقص الركن٨ × ٦ − ٣ × ٢ = ٤٢ م²

مثال: نقسّم ونجمع

📌 مستطيلان (أ) و (ب)
الشكل
(أ)٣ × ٤ = ١٢ سم²
(ب)(٩ − ٤) × ٢ = ١٠ سم²
نجمع١٢ + ١٠ = ٢٢ سم²

معاً: أكمل

📌 تمرين ١ (أ): نقسّم، ثم نوجد مساحة كل جزء، ثم نجمع
الشكل
(أ)٥ × ٤ = ٢٠ سم²
(ب)١١ × ٢ = ٢٢ سم²
نجمع٢٠ + ٢٢ = ٤٢ سم²

معاً: الطول المفقود أولاً

📌 تمرين ٣ (أ)
الشكل
١) الطول المفقود١١ − ٨ = ٣ سم
٢) المستطيلان٨ × ٧ + ٣ × ٤
٣) المساحة٥٦ + ١٢ = ٦٨ سم²

مستطيلٌ ومثلث

📌 تمرين ٤ (ب)
الشكل
المستطيل٨ × ٥ = ٤٠ سم²
المثلث١٢ × ٥ × (١٤ − ٨) = ١٥ سم²
نجمع٤٠ + ١٥ = ٥٥ سم²

مثال: نطرح التجويف

📌 دائرةٌ نصف قطرها ٥ سم، فيها مثلثٌ مقطوع
الشكل
الدائرة٣٫١٤ × ٥² = ٧٨٫٥ سم²
المثلث١٢ × ٤ × ٧ = ١٤ سم²
نطرح٧٨٫٥ − ١٤ = ٦٤٫٥ سم²

انتبه: أخطاءٌ شائعة

📌 التجويف نطرحه◆لا ننسى جزءاً◆نوجد الطول المفقود
✘ جمع التجويف٧٨٫٥ + ١٤
✘ جزءٌ واحدمساحة (أ) فقط = ١٢
✘ الطول الخطأ٩ × ٢

معاً: التجويف

📌 مستطيلٌ ١٥ سم × ١٠ سم فيه فتحة ٦ سم × ٣ سم
الشكل
الكامل١٥ × ١٠ = ١٥٠
نطرح١٥٠ − ١٨ = ١٣٢ سم²

معاً: مثلثٌ فيه دائرة

📌 تمرين ٦ (ب): قطر الدائرة ٢ سم
الشكل
المثلث١٢ × ١٢ × ٩ = ٥٤ سم²
الدائرة٣٫١٤ × ١² = ٣٫١٤ سم²
نطرح٥٤ − ٣٫١٤ = ٥٠٫٨٦ سم²

حديقةٌ ونافورة

📌 حديقةٌ ١٢ م × ٨ م في وسطها نافورةٌ نصف قطرها ٢ م
الحديقة
كاملة١٢ × ٨ = ٩٦ م²
النافورة٣٫١٤ × ٢² = ١٢٫٥٦ م²
العشب٩٦ − ١٢٫٥٦ = ٨٣٫٤٤ م²

شكلٌ ونصف دائرة

📌 مستطيلٌ ١٠ سم × ٥ سم فوقه نصف دائرةٍ قطرها ١٠ سم
المستطيل١٠ × ٥ = ٥٠
نصف الدائرة١٢ × ٣٫١٤ × ٥² = ٣٩٫٢٥
نجمع٥٠ + ٣٩٫٢٥ = ٨٩٫٢٥ سم²

دورك الآن

📌 أوقف الفيديو وأوجد المساحة، ثم تأكّد
السؤال ١٣٠ × ١٠ فيه فتحة ٨ × ٤ (سم)
الحل٣٠٠ − ٣٢ = ٢٦٨ سم²
السؤال ٢١٠ × ٨ ومعه مثلث ٨ و ٦ (سم)
الحل٨٠ + ٢٤ = ١٠٤ سم²

تذكّروا

نقسّم إلى أشكالٍ بسيطة◆نوجد الطول المفقود◆نجمع الأجزاء ونطرح التجويف
