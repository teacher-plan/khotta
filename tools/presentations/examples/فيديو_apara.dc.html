مساحة متوازي الأضلاع وشبه المنحرف

نقصّ ونركّب◆م = ق ع◆م = ½ (ق١ + ق٢) ع

متوازي الأضلاع

📌 نقصّ المثلث من طرفٍ وننقله إلى الطرف الآخر
نقصّ وننقل
المستطيلله القاعدة نفسها والارتفاع نفسه
الصيغةم = ق × ع

مثال: متوازي أضلاع

📌 القاعدة ٧ سم والارتفاع ٥ سم
الشكل
١) الصيغةم = ق ع
٢) نعوّضم = ٧ × ٥ = ٣٥ سم²

شبه المنحرف

📌 نسختان منه تكوّنان متوازي أضلاع
نسختان
قاعدتهق١ + ق٢
الصيغةم = ١٢ (ق١ + ق٢) ع

مثال: ساقية الفلج

📌 مقطعٌ شبه منحرف: ٦٠ سم في الأعلى، ٤٠ سم في القاع، والعمق ٣٠ سم
المقطع
١) نجمع٦٠ + ٤٠ = ١٠٠
٢) النصف١٢ × ١٠٠ = ٥٠
٣) × الارتفاع٥٠ × ٣٠ = ١٥٠٠ سم²

انتبه: أخطاءٌ شائعة

📌 القاعدتان ٤ و ٦ ، والارتفاع ٥
✘ ضرب القاعدتين١٢ × (٤ × ٦) × ٥
✘ بلا نصف(٤ + ٦) × ٥ = ٥٠
✘ نصّف مرتين(٢ + ٣) × ٢٫٥
✔١٢ × (٤ + ٦) × ٥ = ٢٥

معاً: المساحة

📌 م = ق ع◆م = ½ (ق١ + ق٢) ع
متوازي أضلاع ٢٦ ملم و ١٥ ملم٢٦ × ١٥ = ٣٩٠ ملم²
شبه منحرف ٣ و ٨ ، ع = ٧١٢ × ١١ × ٧ = ٣٨٫٥ سم²
شبه منحرف ٨٫٦ و ٥٫٤ ، ع = ٣٫٦١٢ × ١٤ × ٣٫٦ = ٢٥٫٢ سم²

الارتفاع المجهول

📌 مساحته ١٥٠٠ ملم² ، وضلعاه المتوازيان ٤٫٨ سم و ٥٫٢ سم
١) نوحّد٤٨ ملم ، ٥٢ ملم
٢) نعوّض١٥٠٠ = ١٢ × (٤٨ + ٥٢) × ع = ٥٠ ع
٣) نقسمع = ١٥٠٠ ÷ ٥٠ = ٣٠ ملم

مسألة: أرض المزرعة

📌 قطعة أرضٍ شبه منحرف: ضلعاها المتوازيان ٤٥ م و ٣٥ م، والمسافة بينهما ٢٠ م
١) نجمع٤٥ + ٣٥ = ٨٠
٢) النصف١٢ × ٨٠ = ٤٠
٣) × الارتفاع٤٠ × ٢٠ = ٨٠٠ م²
نتحقّقبين ٣٥ × ٢٠ = ٧٠٠ و ٤٥ × ٢٠ = ٩٠٠

القاعدة المجهولة

📌 متوازي أضلاع مساحته ٩٦ سم² وارتفاعه ٨ سم
١) نعوّض٩٦ = ق × ٨
٢) نقسمق = ٩٦ ÷ ٨ = ١٢ سم

دورك الآن

📌 أوقف الفيديو وأوجد القاعدة، ثم تأكّد
السؤالمتوازي أضلاع: م = ٨٣٢ ملم² ، ع = ٢٦ ملم
الحلق = ٨٣٢ ÷ ٢٦ = ٣٢ ملم
