مساحة متوازي الأضلاع وشبه المنحرف

نقصّ ونركّب◆م = ق ع◆م = ½ (ق١ + ق٢) ع

متوازي الأضلاع

📌 نقصّ المثلث من طرفٍ وننقله إلى الطرف الآخر
نقصّ وننقل
المستطيلله القاعدة نفسها والارتفاع نفسه
الصيغةم = ق × ع

مثال: متوازي أضلاع

📌 القاعدة ٧ سم والارتفاع ٥ سم
الشكل
١) الصيغةم = ق ع
٢) نعوّضم = ٧ × ٥ = ٣٥ سم²

انتبه: الارتفاع عمودي، لا مائل

📌 القاعدة ٨ سم ، الارتفاع ٤ سم ، والضلع المائل ٥ سم
الشكل
✘٨ × ٥ = ٤٠ سم²
✔٨ × ٤ = ٣٢ سم²

شبه المنحرف

📌 نسختان منه تكوّنان متوازي أضلاع
نسختان
قاعدتهق١ + ق٢
الصيغةم = ١٢ (ق١ + ق٢) ع

مثال ٧-٤: شبه منحرف

📌 الضلعان المتوازيان ١٢ ملم و ١٨ ملم ، والارتفاع ٨ ملم
الشكل
١) نجمع١٢ + ١٨ = ٣٠
٢) النصف١٢ × ٣٠ = ١٥
٣) × الارتفاع١٥ × ٨ = ١٢٠ ملم²

مثال: ساقية الفلج

📌 مقطعٌ شبه منحرف: ٦٠ سم في الأعلى، ٤٠ سم في القاع، والعمق ٣٠ سم
المقطع
١) نجمع٦٠ + ٤٠ = ١٠٠
٢) النصف١٢ × ١٠٠ = ٥٠
٣) × الارتفاع٥٠ × ٣٠ = ١٥٠٠ سم²

انتبه: أخطاءٌ شائعة

📌 القاعدتان ٤ و ٦ ، والارتفاع ٥
✘ ضرب القاعدتين١٢ × (٤ × ٦) × ٥
✘ بلا نصف(٤ + ٦) × ٥ = ٥٠
✘ نصّف مرتين(٢ + ٣) × ٢٫٥
✔١٢ × (٤ + ٦) × ٥ = ٢٥

معاً: المساحة

📌 م = ق ع◆م = ½ (ق١ + ق٢) ع
متوازي أضلاع ٢٦ ملم و ١٥ ملم٢٦ × ١٥ = ٣٩٠ ملم²
شبه منحرف ٣ و ٨ ، ع = ٧١٢ × ١١ × ٧ = ٣٨٫٥ سم²
شبه منحرف ٨٫٦ و ٥٫٤ ، ع = ٣٫٦١٢ × ١٤ × ٣٫٦ = ٢٥٫٢ سم²

الارتفاع المجهول

📌 مساحته ١٥٠٠ ملم² ، وضلعاه المتوازيان ٤٫٨ سم و ٥٫٢ سم
١) نوحّد٤٨ ملم ، ٥٢ ملم
٢) نعوّض١٥٠٠ = ١٢ × (٤٨ + ٥٢) × ع = ٥٠ ع
٣) نقسمع = ١٥٠٠ ÷ ٥٠ = ٣٠ ملم

القاعدة المجهولة

📌 متوازي أضلاع مساحته ٤٣٫٤ سم² وارتفاعه ٢٨ ملم
١) نوحّد٤٣٫٤ سم² = ٤٣٤٠ ملم²
٢) نقسمق = ٤٣٤٠ ÷ ٢٨ = ١٥٥ ملم
بالسنتيمتر١٥٫٥ سم

واجب عائشة

📌 المساحة المظلّلة = مساحة الشكل (ب) − مساحة الشكل (أ)
الواجب
الخطأأبعاد (أ) بالمليمتر، و(ب) بالسنتيمتر
١) نحوّل١٨٠ ملم² = ١٫٨ سم²
٢) نطرح٧٢ − ١٫٨ = ٧٠٫٢ سم²

نتحقّق بالتقريب

📌 نقرّب الأبعاد لأعدادٍ كاملة، ثم نختار البطاقة الأقرب
شبه منحرف١٢ × (٥ + ٥) × ٤ = ٢٠ ← ١٨٫٨١ سم²
متوازي أضلاع٣ × ٣ = ٩ ← ٩٫٨٦ سم²

دورك الآن

📌 أوقف الفيديو وأوجد القاعدة، ثم تأكّد
